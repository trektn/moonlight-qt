cmake_minimum_required(VERSION 3.20)
project(MicBridgeReceiver LANGUAGES CXX)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_executable(micbridge-receiver receiver.cpp)
target_compile_features(micbridge-receiver PRIVATE cxx_std_17)
target_compile_definitions(micbridge-receiver PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0601)
target_link_libraries(micbridge-receiver PRIVATE ws2_32 ole32 uuid avrt)
if(MINGW)
  target_link_options(micbridge-receiver PRIVATE -static -municode)
endif()
